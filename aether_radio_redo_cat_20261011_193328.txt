===== README.md =====
# Aether FM

Aether FM is a web radio station player with a tactile hardware-inspired interface.
It organizes online live audio transmissions across diverse genres and frequency bands.
The player supports native audio streams as well as adaptive HLS live broadcast feeds.
Users can discover stations, browse curated genres, and tune into high-fidelity streams.
Features include a spectrogram equalizer, custom frequency dial, and favorite presets.
Station metadata is cached locally for fast startup with background freshness updates.
The responsive interface adapts seamlessly from compact mobile viewports to wide screens.
Built with React 19, Vite, Tailwind CSS, and Lucide icons for crisp typography and design.
Run `npm run dev` to start the local development server or `npm run build` to bundle.
Enjoy continuous, low-latency live radio broadcasts with precise audio controls.

import { useState, useEffect, useCallback } from "react";

import { useSpeedTest } from "../hooks/useSpeedTest";

const STATUS_GREEN = "#00ff41";
const STATUS_YELLOW = "#facc15";
const STATUS_RED = "#ef4444";
const STATUS_GRAY = "#666666";

function getSpeedColor(mbps) {
  if (mbps === null || mbps === undefined) return STATUS_GRAY;
  if (mbps >= 20) return STATUS_GREEN;
  if (mbps >= 5) return STATUS_YELLOW;
  return STATUS_RED;
}

function getNetworkInfo() {
  const isOnline = typeof navigator !== "undefined" ? navigator.onLine : true;
  const connection =
    typeof navigator !== "undefined"
      ? navigator.connection ||
        navigator.mozConnection ||
        navigator.webkitConnection
      : null;

  if (!isOnline) {
    return {
      level: 1,
      color: STATUS_GRAY,
      speedText: "Offline",
      subText: "No Connection",
      downlink: 0,
      effectiveType: "offline",
      online: false,
    };
  }

  if (!connection || connection.downlink === undefined) {
    return {
      level: 4,
      color: STATUS_GREEN,
      speedText: "Online",
      subText: "Active Network",
      downlink: null,
      effectiveType: null,
      online: true,
    };
  }

  const downlink = connection.downlink;
  const effectiveType = connection.effectiveType
    ? connection.effectiveType.toUpperCase()
    : "NETWORK";

  let level;
  let color;

  if (downlink >= 20) {
    level = 4;
    color = STATUS_GREEN;
  } else if (downlink >= 5) {
    level = 3;
    color = STATUS_YELLOW;
  } else {
    level = 2;
    color = STATUS_RED;
  }

  return {
    level,
    color,
    speedText: `${downlink.toFixed(1)} Mbps`,
    subText: effectiveType,
    downlink,
    effectiveType,
    online: true,
  };
}

function isSameNetwork(a, b) {
  if (a === b) return true;
  if (!a || !b) return false;
  return (
    a.online === b.online &&
    a.level === b.level &&
    a.color === b.color &&
    a.speedText === b.speedText &&
    a.subText === b.subText &&
    a.downlink === b.downlink &&
    a.effectiveType === b.effectiveType
  );
}

export default function NetworkIndicator() {
  const [network, setNetwork] = useState(getNetworkInfo);
  const [showModal, setShowModal] = useState(false);

  const {
    run: runSpeedTest,
    result: testResult,
    status: testStatus,
    error: testError,
    abort: abortSpeedTest,
  } = useSpeedTest();

  const updateNetwork = useCallback(() => {
    const next = getNetworkInfo();
    setNetwork((prev) => (isSameNetwork(prev, next) ? prev : next));
  }, []);

  useEffect(() => {
    const handleVisibilityChange = () => {
      if (document.visibilityState === "hidden") {
        abortSpeedTest();
      } else {
        updateNetwork();
      }
    };

    const connection =
      typeof navigator !== "undefined"
        ? navigator.connection ||
          navigator.mozConnection ||
          navigator.webkitConnection
        : null;

    window.addEventListener("online", updateNetwork);
    window.addEventListener("offline", updateNetwork);
    document.addEventListener("visibilitychange", handleVisibilityChange);

    if (connection && typeof connection.addEventListener === "function") {
      connection.addEventListener("change", updateNetwork);
    }

    return () => {
      abortSpeedTest();
      window.removeEventListener("online", updateNetwork);
      window.removeEventListener("offline", updateNetwork);
      document.removeEventListener("visibilitychange", handleVisibilityChange);
      if (connection && typeof connection.removeEventListener === "function") {
        connection.removeEventListener("change", updateNetwork);
      }
    };
  }, [updateNetwork, abortSpeedTest]);

  const handleOpenModal = () => {
    setShowModal(true);
    runSpeedTest();
  };

  const handleCloseModal = () => {
    setShowModal(false);
    abortSpeedTest();
  };

  const { level, color, speedText, subText, online } = network;

  // Render measured value
  const renderMeasuredText = () => {
    if (testStatus === "running") {
      return <span className="text-[#facc15] font-bold">Testing…</span>;
    }
    if (testStatus === "done" && testResult) {
      return (
        <span
          className="font-bold"
          style={{ color: getSpeedColor(testResult.mbps) }}
        >
          {testResult.mbps.toFixed(1)} Mbps
        </span>
      );
    }
    if (testStatus === "error") {
      return <span className="text-red-400 font-bold">Test failed</span>;
    }
    return <span className="text-white/40">—</span>;
  };

  const modalContent = showModal ? (
    <div
      className="fixed inset-0 z-[100] flex items-center justify-center bg-black/80 p-4 backdrop-blur-sm sm:hidden"
      onClick={handleCloseModal}
    >
      <div
        className="w-full max-w-xs border border-white/20 bg-[#111] p-5 shadow-2xl"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="flex items-center justify-between border-b border-white/10 pb-3 mb-4">
          <div className="flex items-center gap-2">
            {online ? (
              <Wifi className="h-4 w-4" style={{ color }} />
            ) : (
              <WifiOff className="h-4 w-4" style={{ color }} />
            )}
            <span className="text-xs font-bold uppercase tracking-[0.2em] text-white">
              Network Link
            </span>
          </div>
          <button
            onClick={handleCloseModal}
            className="text-white/40 hover:text-white transition-colors"
            aria-label="Close"
          >
            <X className="h-4 w-4" />
          </button>
        </div>

        <div className="space-y-3 py-2 text-center">
          <div className="flex justify-center items-end gap-1 h-6">
            <span
              className="w-1.5 h-2"
              style={{
                backgroundColor:
                  level >= 1 ? color : "rgba(255,255,255,0.15)",
              }}
            />
            <span
              className="w-1.5 h-3"
              style={{
                backgroundColor:
                  level >= 2 ? color : "rgba(255,255,255,0.15)",
              }}
            />
            <span
              className="w-1.5 h-[18px]"
              style={{
                backgroundColor:
                  level >= 3 ? color : "rgba(255,255,255,0.15)",
              }}
            />
            <span
              className="w-1.5 h-6"
              style={{
                backgroundColor:
                  level >= 4 ? color : "rgba(255,255,255,0.15)",
              }}
            />
          </div>

          <div className="space-y-2 py-1 text-left">

import { useCallback, useEffect, useRef, useState } from "react";
import Hls from "hls.js";
import { PlayerContext } from "./PlayerContextDef";

// ---------------------------------------------------------------------------
// Switches / helpers
// ---------------------------------------------------------------------------

// Set to false to go back to plain <audio> playback with no Web Audio at all.
const ANALYSER_ENABLED = true;

const FAVORITES_KEY = "aether_favorites";
const VOLUME_KEY = "aether_volume";
const MUTED_KEY = "aether_muted";
const LAST_STATION_KEY = "aether_last_station";

// url -> false once we learn the server sends no CORS headers.
// Such streams still play, but the browser will not let us analyse them.
const corsSupport = new Map();

function loadFavorites() {
  try {
    const raw = localStorage.getItem(FAVORITES_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
}

function saveFavorites(favs) {
  try {
    localStorage.setItem(FAVORITES_KEY, JSON.stringify(favs));
  } catch {
    /* quota / private-mode - ignore */
  }
}

function loadVolume() {
  try {
    const raw = localStorage.getItem(VOLUME_KEY);
    if (raw === null) return 0.8;
    const n = Number(raw);
    return Number.isFinite(n) ? Math.max(0, Math.min(1, n)) : 0.8;
  } catch {
    return 0.8;
  }
}

function loadMuted() {
  try {
    return localStorage.getItem(MUTED_KEY) === "1";
  } catch {
    return false;
  }
}

function loadLastStation() {
  try {
    const raw = localStorage.getItem(LAST_STATION_KEY);
    if (!raw) return null;
    const parsed = JSON.parse(raw);
    // Minimal shape check — anything malformed is discarded.
    if (
      parsed &&
      typeof parsed === "object" &&
      typeof parsed.streamUrl === "string" &&
      parsed.streamUrl.length > 0
    ) {
      return parsed;
    }
  } catch {
    /* ignore */
  }
  return null;
}

function saveLastStation(station) {
  try {
    if (!station || !station.streamUrl) {
      localStorage.removeItem(LAST_STATION_KEY);
      return;
    }
    localStorage.setItem(LAST_STATION_KEY, JSON.stringify(station));
  } catch {
    /* quota / private-mode - ignore */
  }
}

function stationKey(station) {
  return station?._id || station?.streamUrl || "";
}

function needsHls(url) {
  if (!url) return false;
  if (!url.includes(".m3u8")) return false;
  const audio = document.createElement("audio");
  return !audio.canPlayType("application/vnd.apple.mpegurl");
}

function deriveFormat(url) {
  if (!url) return "LIVE STREAM";
  const lower = url.toLowerCase();
  if (lower.includes(".m3u8")) return "HLS LIVE STREAM";
  if (lower.includes(".mp3")) return "MP3 LIVE STREAM";
  if (lower.includes(".aac")) return "AAC LIVE STREAM";
  if (lower.includes(".ogg")) return "OGG LIVE STREAM";
  return "LIVE STREAM";
}

// ---------------------------------------------------------------------------
// Provider
// ---------------------------------------------------------------------------

export function PlayerProvider({ children }) {
  const audioRef = useRef(null);
  const unbindRef = useRef(null);
  const hlsRef = useRef(null);
  const stoppingRef = useRef(false);
  const hlsMediaRecoveredRef = useRef(false);
  const hlsNetworkRecoveredRef = useRef(false);

  // Web Audio
  const ctxRef = useRef(null);
  const analyserNodeRef = useRef(null);
  const sourceRef = useRef(null);
  // Exposed to the UI. Non-null ONLY while the current <audio> element is
  // routed through the analyser (i.e. real audio data is available).
  const analyserRef = useRef(null);

  // Playback bookkeeping.
  // Volume / mute refs read the same persisted values as the state below, so
  // the very first <audio> element picks up the restored settings.
  const tryingCorsRef = useRef(false);
  const stationRef = useRef(null);
  const retryRef = useRef(null);
  const volumeRef = useRef(loadVolume());
  const mutedRef = useRef(loadMuted());
  const stallTimerRef = useRef(null);
  const stallRetryCountRef = useRef(0);

  // The last station is restored paused — not played. Browsers block autoplay
  // with sound until a user gesture, so trying to play on mount would just
  // fail silently and confuse the status badge.
  const [current, setCurrent] = useState(loadLastStation);
  const [isPlaying, setIsPlaying] = useState(false);
  const [volume, setVolumeState] = useState(loadVolume);
  const [muted, setMuted] = useState(loadMuted);
  const [error, setError] = useState(null);
  const [status, setStatus] = useState("idle"); // idle | connecting | connected | error
  const [playlist, setPlaylist] = useState([]);
  const [index, setIndex] = useState(-1);
  const [favorites, setFavorites] = useState(loadFavorites);

  const clearStallTimer = useCallback(() => {
    if (stallTimerRef.current) {
      clearTimeout(stallTimerRef.current);
      stallTimerRef.current = null;
    }
  }, []);

  const handleStallTimeout = useCallback(() => {
    clearStallTimer();
    const station = stationRef.current || current;
    if (!station) return;
    if (stallRetryCountRef.current === 0) {
      stallRetryCountRef.current = 1;
      if (retryRef.current) {
        retryRef.current(station);
      }
    } else {
      setIsPlaying(false);
      setError("Stream stalled");
      setStatus("error");
    }
  }, [clearStallTimer, current]);

  // Persist favorites
  useEffect(() => {
    saveFavorites(favorites);
  }, [favorites]);

  // Persist volume
  useEffect(() => {
    try {
      localStorage.setItem(VOLUME_KEY, String(volume));
    } catch {
      /* ignore */
    }
  }, [volume]);

  // Persist muted
  useEffect(() => {
    try {
      localStorage.setItem(MUTED_KEY, muted ? "1" : "0");
    } catch {
      /* ignore */
    }
  }, [muted]);

  // Persist last station so reopening the app returns you to it, paused.
  useEffect(() => {
    saveLastStation(current);
  }, [current]);

  const destroyHls = useCallback(() => {
    if (hlsRef.current) {
      hlsRef.current.destroy();
      hlsRef.current = null;
    }
  }, []);

  // If an attempt made with crossorigin="anonymous" fails, the server almost
  // certainly lacks CORS headers. Remember that and retry as a plain element.
  const handleCorsFailure = useCallback(() => {
    if (!tryingCorsRef.current) return false;
    tryingCorsRef.current = false;
    const station = stationRef.current;
    if (!station) return false;
    corsSupport.set(station.streamUrl, false);
    if (retryRef.current) retryRef.current(station);
    return true;
  }, []);

  // Wire the listeners for one <audio> element. Returns an unbind function.
  const bindAudio = useCallback(
    (audio) => {
      const onPlay = () => setIsPlaying(true);

      const onPause = () => {
        clearStallTimer();
        setIsPlaying(false);
        setStatus((prev) => (prev === "error" ? "error" : "idle"));
      };

      const onPlaying = () => {
        clearStallTimer();
        stallRetryCountRef.current = 0;
        tryingCorsRef.current = false;
        const ctx = ctxRef.current;
        if (ctx && ctx.state === "suspended") ctx.resume().catch(() => {});
        setStatus("connected");
      };

      const onStall = () => {
        setStatus("connecting");
        if (!stallTimerRef.current) {
          stallTimerRef.current = setTimeout(handleStallTimeout, 15000);
        }
      };

      const onWaiting = onStall;
      const onStalled = onStall;

      const onEnded = () => {
        if (stoppingRef.current) return;
        handleStallTimeout();
      };

      const onError = () => {
        clearStallTimer();
        if (stoppingRef.current) return;
        if (handleCorsFailure()) return;
        setIsPlaying(false);
        setError("Stream failed to load");
        setStatus("error");
      };

      audio.addEventListener("play", onPlay);
      audio.addEventListener("pause", onPause);
      audio.addEventListener("playing", onPlaying);
      audio.addEventListener("waiting", onWaiting);
      audio.addEventListener("stalled", onStalled);
      audio.addEventListener("ended", onEnded);
      audio.addEventListener("error", onError);

      return () => {
        audio.removeEventListener("play", onPlay);
        audio.removeEventListener("pause", onPause);
        audio.removeEventListener("playing", onPlaying);
        audio.removeEventListener("waiting", onWaiting);
        audio.removeEventListener("stalled", onStalled);
        audio.removeEventListener("ended", onEnded);
        audio.removeEventListener("error", onError);
      };
    },
    [clearStallTimer, handleCorsFailure, handleStallTimeout]
  );

  // A media element that has been routed into Web Audio cannot be un-routed,
  // so every playback start gets a fresh element.
  const makeAudio = useCallback(() => {
    if (unbindRef.current) {
      unbindRef.current();
      unbindRef.current = null;
    }
    try {
      if (sourceRef.current) sourceRef.current.disconnect();
    } catch {
      /* ignore */
    }
    sourceRef.current = null;
    analyserRef.current = null;

    const old = audioRef.current;
    if (old) {
      old.pause();
      old.removeAttribute("src");
      old.load();
    }

    const audio = new Audio();
    audio.preload = "none";
    audio.volume = volumeRef.current;
    audio.muted = mutedRef.current;
    audioRef.current = audio;
    unbindRef.current = bindAudio(audio);
    return audio;
  }, [bindAudio]);

  const connectAnalyser = useCallback((audio) => {
    try {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (!AudioCtx) return;

      if (!ctxRef.current) ctxRef.current = new AudioCtx();
      const ctx = ctxRef.current;
      if (ctx.state === "suspended") ctx.resume().catch(() => {});

      if (!analyserNodeRef.current) {
        const node = ctx.createAnalyser();
        node.fftSize = 512;
        node.smoothingTimeConstant = 0.7;
        node.minDecibels = -90;
        node.maxDecibels = -25;
        node.connect(ctx.destination);
        analyserNodeRef.current = node;
      }

      const source = ctx.createMediaElementSource(audio);
      source.connect(analyserNodeRef.current);
      sourceRef.current = source;
      analyserRef.current = analyserNodeRef.current;
    } catch (err) {
      console.warn("Analyser unavailable:", err);
      analyserRef.current = null;
    }
  }, []);

  const startPlayback = useCallback(
    (station) => {
      clearStallTimer();
      const url = station.streamUrl;

      stoppingRef.current = false;
      stationRef.current = station;
      destroyHls();
      hlsMediaRecoveredRef.current = false;
      hlsNetworkRecoveredRef.current = false;

      const viaHls = needsHls(url) && Hls.isSupported();
      // hls.js feeds the element a same-origin blob, so it can always be
      // analysed. Plain streams need CORS headers, so we try and fall back.
      const wantCors =
        ANALYSER_ENABLED && !viaHls && corsSupport.get(url) !== false;

      const audio = makeAudio();
      tryingCorsRef.current = wantCors;
      if (wantCors) audio.crossOrigin = "anonymous";
      if (ANALYSER_ENABLED && (wantCors || viaHls)) connectAnalyser(audio);

      if (viaHls) {
        const hls = new Hls();
        hls.loadSource(url);
        hls.attachMedia(audio);
        hls.on(Hls.Events.ERROR, (_event, data) => {
          if (data.fatal) {
            if (
              data.type === Hls.ErrorTypes.MEDIA_ERROR &&
              !hlsMediaRecoveredRef.current
            ) {
              hlsMediaRecoveredRef.current = true;
              hls.recoverMediaError();
              return;
            }
            if (
              data.type === Hls.ErrorTypes.NETWORK_ERROR &&
              !hlsNetworkRecoveredRef.current
            ) {
              hlsNetworkRecoveredRef.current = true;
              hls.startLoad();
              return;
            }
            destroyHls();
            setError("HLS stream error");
            setStatus("error");
            setIsPlaying(false);
          }
        });
        hlsRef.current = hls;
      } else {
        audio.src = url;
      }

      audio.play().catch((err) => {
        if (err?.name === "AbortError") return;
        // A newer element has taken over; this failure is stale.
        if (audioRef.current !== audio) return;
        if (handleCorsFailure()) return;
        console.error("Audio playback error:", err);
        setIsPlaying(false);
        setError("Stream failed to load");
        setStatus("error");
      });
    },
    [clearStallTimer, connectAnalyser, destroyHls, handleCorsFailure, makeAudio]
  );

  useEffect(() => {
    retryRef.current = startPlayback;
  }, [startPlayback]);

  // Unmount cleanup
  useEffect(() => {
    return () => {
      clearStallTimer();
      if (unbindRef.current) unbindRef.current();
      if (audioRef.current) audioRef.current.pause();
    };
  }, [clearStallTimer]);

  const isCurrent = useCallback(
    (station) => {
      if (!current || !station) return false;
      return stationKey(current) === stationKey(station);
    },
    [current]
  );

  const play = useCallback(
    (station, list) => {
      if (!station || !station.streamUrl) return;

      const alreadyCurrent =
        current && stationKey(current) === stationKey(station);

      // Already current and healthy: skip re-init, only refresh the playlist.
      if (alreadyCurrent && audioRef.current && status !== "error") {
        if (Array.isArray(list) && list.length > 0) {
          const newIdx = list.findIndex(
            (s) => stationKey(s) === stationKey(station)
          );
          setPlaylist(list);
          setIndex(newIdx >= 0 ? newIdx : 0);
        }
        return;
      }

      setError(null);
      setCurrent(station);
      setStatus("connecting");

      if (Array.isArray(list) && list.length > 0) {
        setPlaylist(list);
        const idx = list.findIndex(
          (s) => stationKey(s) === stationKey(station)
        );
        setIndex(idx >= 0 ? idx : 0);
      } else {
        setPlaylist((prev) => (prev.length > 0 ? prev : [station]));
        setIndex(0);
      }

      startPlayback(station);
    },
    [current, startPlayback, status]
  );

  const toggle = useCallback(() => {
    const audio = audioRef.current;
    if (!current) return;

    if (isPlaying) {
      if (audio) audio.pause();
    } else {
      // If we restored a station on mount, audioRef is null until the first
      // play. startPlayback creates the element and starts it.
      setError(null);
      setStatus("connecting");
      startPlayback(current);
    }
  }, [current, isPlaying, startPlayback]);

  const stop = useCallback(() => {
    clearStallTimer();
    stoppingRef.current = true;
    tryingCorsRef.current = false;
    destroyHls();
    const audio = audioRef.current;
    if (audio) {
      audio.pause();
      audio.src = "";
    }
    analyserRef.current = null;
    setCurrent(null);
    setIsPlaying(false);
    setError(null);
    setStatus("idle");
    setPlaylist([]);
    setIndex(-1);
    try {
      localStorage.removeItem(LAST_STATION_KEY);
    } catch {
      /* ignore */
    }
  }, [clearStallTimer, destroyHls]);

  const next = useCallback(() => {
    if (playlist.length === 0) return;
    const nextIdx = (index + 1) % playlist.length;
    play(playlist[nextIdx], playlist);
  }, [index, play, playlist]);

  const prev = useCallback(() => {
    if (playlist.length === 0) return;
    const prevIdx = (index - 1 + playlist.length) % playlist.length;
    play(playlist[prevIdx], playlist);
  }, [index, play, playlist]);

  const handleSetVolume = useCallback((val) => {
    const newVol = Math.max(0, Math.min(1, val));
    volumeRef.current = newVol;
    setVolumeState(newVol);
    if (audioRef.current) audioRef.current.volume = newVol;
  }, []);

  const toggleMute = useCallback(() => {
    const nextMuted = !mutedRef.current;
    mutedRef.current = nextMuted;
    setMuted(nextMuted);
    if (audioRef.current) audioRef.current.muted = nextMuted;
  }, []);

  const isFavorite = useCallback(
    (station) => {
      const key = stationKey(station);
      return key ? favorites.includes(key) : false;
    },
    [favorites]
  );

  const toggleFavorite = useCallback((station) => {
    const key = stationKey(station);
    if (!key) return;
    setFavorites((prev) =>
      prev.includes(key) ? prev.filter((k) => k !== key) : [...prev, key]
    );
  }, []);

  const format = deriveFormat(current?.streamUrl);

  const msRef = useRef({});
  const canSkip = playlist.length > 1;

  useEffect(() => {
    msRef.current = {
      play: () => {
        if (!isPlaying) toggle();
      },
      pause: () => {
        if (isPlaying) toggle();
      },
      stop,
      previoustrack: prev,
      nexttrack: next,
    };
  }, [isPlaying, toggle, stop, prev, next]);

  useEffect(() => {
    if (!("mediaSession" in navigator)) return;
    const ms = navigator.mediaSession;
    const live = ["play", "pause", "stop"];
    const skip = ["previoustrack", "nexttrack"];
    for (const a of [...live, ...skip]) {
      const on = live.includes(a) || canSkip;
      try {
        ms.setActionHandler(a, on ? () => msRef.current[a]?.() : null);
      } catch {
        /* unsupported */
      }
    }
  }, [canSkip]);

  useEffect(() => {
    if (!("mediaSession" in navigator)) return;
    navigator.mediaSession.metadata = current
      ? new MediaMetadata({
          title: current.name,
          artist: current.genre || "Live radio",
          album: "Aether FM",
        })
      : null;
  }, [current]);

  useEffect(() => {
    if (!("mediaSession" in navigator)) return;
    navigator.mediaSession.playbackState = !current
      ? "none"
      : isPlaying
      ? "playing"
      : "paused";
  }, [current, isPlaying]);

  return (
    <PlayerContext.Provider
      value={{
        current,
        isPlaying,
        volume,
        muted,
        error,
        status,
        format,
        playlist,
        index,
        play,
        toggle,
        stop,
        next,
        prev,
        setVolume: handleSetVolume,
        toggleMute,
        isCurrent,
        isFavorite,
        toggleFavorite,
        audioRef,
        analyserRef,
      }}
    >
      {children}
    </PlayerContext.Provider>
  );
}

import { useCallback, useEffect, useState } from "react";
import { StationsContext } from "./StationsContextDef";

const API = "https://laughing-barnacle-ashn.onrender.com/stations";
const CACHE_KEY = "aether:stations:v2";
const CACHE_TTL_MS = 6 * 60 * 60 * 1000; // 6 hours

// ---------------------------------------------------------------------------
// Normalisation helpers
// ---------------------------------------------------------------------------

function normalizeStation(raw) {
  if (!raw || typeof raw !== "object") return null;
  const streamUrl = raw.streamUrl ?? raw.url ?? raw.stream;
  if (!streamUrl) return null;
  const name = raw.name ?? raw.title ?? "Untitled";
  const genre =
    typeof raw.genre === "string" && raw.genre.trim()
      ? raw.genre.trim()
      : "Unknown";

  return {
    ...raw,
    name,
    streamUrl,
    genre,
  };
}

function normalizeStations(list) {
  if (!Array.isArray(list)) return [];
  return list.map(normalizeStation).filter(Boolean);
}

// ---------------------------------------------------------------------------
// localStorage helpers – every access wrapped in try/catch
// ---------------------------------------------------------------------------

function readCache() {
  try {
    const raw = localStorage.getItem(CACHE_KEY);
    if (!raw) return null;
    const parsed = JSON.parse(raw);
    if (
      parsed &&
      typeof parsed === "object" &&
      typeof parsed.ts === "number" &&
      Array.isArray(parsed.data) &&
      parsed.data.length > 0
    ) {
      const normalized = normalizeStations(parsed.data);
      if (normalized.length > 0) {
        return { ts: parsed.ts, data: normalized };
      }
    }
    // Invalid shape – discard
    localStorage.removeItem(CACHE_KEY);
  } catch {
    try { localStorage.removeItem(CACHE_KEY); } catch { /* ignore */ }
  }
  return null;
}

function writeCache(data) {
  try {
    localStorage.setItem(
      CACHE_KEY,
      JSON.stringify({ ts: Date.now(), data })
    );
  } catch {
    /* quota exceeded or storage blocked – ignore */
  }
}

// ---------------------------------------------------------------------------
// Lazy initialiser – runs in useState(), not in an effect
// ---------------------------------------------------------------------------

function initFromCache() {
  const entry = readCache();
  if (entry) {
    const fresh = Date.now() - entry.ts < CACHE_TTL_MS;
    return {
      stations: entry.data,
      loading: false,       // we have data → no loading state
      error: null,
      needsFetch: !fresh,   // stale → background refresh
    };
  }
  return {
    stations: [],
    loading: true,          // no data at all → show loading
    error: null,
    needsFetch: true,
  };
}

// ---------------------------------------------------------------------------
// Provider
// ---------------------------------------------------------------------------

export function StationsProvider({ children }) {
  const [state, setState] = useState(() => {
    const initial = initFromCache();
    return {
      stations: initial.stations,
      loading: initial.loading,
      error: initial.error,
      needsFetch: initial.needsFetch,
    };
  });
  const [initialNeedsFetch] = useState(state.needsFetch);

  // Fetch helper – returns the data array or throws
  const fetchStations = useCallback(async () => {
    const res = await fetch(API);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const data = await res.json();
    const list = Array.isArray(data) ? data : data?.stations || [];
    return normalizeStations(list);
  }, []);

  // Initial / stale-refresh effect
  useEffect(() => {
    if (!initialNeedsFetch) return;

    let cancelled = false;

    (async () => {
      try {
        const data = await fetchStations();
        if (cancelled) return;
        writeCache(data);
        setState({ stations: data, loading: false, error: null });
      } catch (err) {
        if (cancelled) return;
        // If we already had cached (stale) data, keep it – no error shown.
        setState((prev) => {
          if (prev.stations.length > 0) {
            return { ...prev, loading: false };
          }
          // No cached data at all – show the error
          return { stations: [], loading: false, error: err.message };
        });
      }
    })();

    return () => {
      cancelled = true;
    };
  }, [initialNeedsFetch, fetchStations]);

  // Manual reload – ignores TTL, always refetches
  const reload = useCallback(() => {
    // Only set loading if we have no data yet
    setState((prev) => ({
      ...prev,
      loading: prev.stations.length === 0,
      error: null,
    }));

    fetchStations()
      .then((data) => {
        writeCache(data);
        setState({ stations: data, loading: false, error: null });
      })
      .catch((err) => {
        setState((prev) => {
          if (prev.stations.length > 0) {
            return { ...prev, loading: false };
          }
          return { stations: [], loading: false, error: err.message };
        });
      });
  }, [fetchStations]);

  const value = {
    stations: state.stations,
    loading: state.loading,
    error: state.error,
    reload,
  };

  return (
    <StationsContext.Provider value={value}>
      {children}
    </StationsContext.Provider>
  );
}

// src/hooks/usePlayerShortcuts.js
import { useEffect } from "react";
import { usePlayer } from "../context/usePlayer";

const TEXT_TAGS = new Set(["INPUT", "TEXTAREA", "SELECT"]);

const isButtonOrA = (t) =>
  Boolean(t && (t.tagName === "BUTTON" || t.tagName === "A" || t.closest?.("button, a")));

export function usePlayerShortcuts() {
  const { current, toggle, next, prev, playlist } = usePlayer();

  useEffect(() => {
    if (!current) return; // nothing loaded: let keys scroll/type normally

    const ignore = (e) => {
      const t = e.target;
      if (e.ctrlKey || e.metaKey || e.altKey) return true;
      if (!t) return false;
      if (t.isContentEditable) return true;
      return TEXT_TAGS.has(t.tagName) && t.type !== "range";
    };

    const onKeyDown = (e) => {
      if (ignore(e)) return;

      // Shift+P → previous
      if (e.shiftKey && e.code === "KeyP") {
        e.preventDefault();
        if (!e.repeat && playlist.length > 1) prev();
        return;
      }

      // Shift+N → next
      if (e.shiftKey && e.code === "KeyN") {
        e.preventDefault();
        if (!e.repeat && playlist.length > 1) next();
        return;
      }

      // Space → play/pause
      if (e.code === "Space") {
        if (isButtonOrA(e.target)) return;
        e.preventDefault();
        if (!e.repeat) toggle();
      }
    };

    const onKeyUp = (e) => {
      if (ignore(e)) return;

      if (e.code === "Space") {
        if (isButtonOrA(e.target)) return;
        e.preventDefault();
        return;
      }

      if ((e.code === "KeyP" || e.code === "KeyN") && e.shiftKey) {
        e.preventDefault();
      }
    };

    window.addEventListener("keydown", onKeyDown);
    window.addEventListener("keyup", onKeyUp);
    return () => {
      window.removeEventListener("keydown", onKeyDown);
      window.removeEventListener("keyup", onKeyUp);
    };
  }, [current, toggle, next, prev, playlist.length]);
}

import { useState, useRef, useCallback, useEffect } from "react";

const DEBOUNCE_MS = 60 * 1000; // 60 seconds

export function useSpeedTest() {
  const [status, setStatus] = useState("idle"); // "idle" | "running" | "done" | "error"
  const [result, setResult] = useState(null);
  const [error, setError] = useState(null);

  const controllerRef = useRef(null);
  const isMountedRef = useRef(true);
  const lastCompletedRef = useRef(0);
  const statusRef = useRef("idle");
  const resultRef = useRef(null);

  useEffect(() => {
    isMountedRef.current = true;
    return () => {
      isMountedRef.current = false;
      if (controllerRef.current) {
        controllerRef.current.abort();
      }
    };
  }, []);

  const abort = useCallback(() => {
    if (controllerRef.current) {
      controllerRef.current.abort();
      controllerRef.current = null;
    }
    if (isMountedRef.current) {
      statusRef.current = resultRef.current ? "done" : "idle";
      setStatus(statusRef.current);
    }
  }, []);

  const reset = useCallback(() => {
    if (controllerRef.current) {
      controllerRef.current.abort();
      controllerRef.current = null;
    }
    lastCompletedRef.current = 0;
    resultRef.current = null;
    if (isMountedRef.current) {
      setResult(null);
      setError(null);
      statusRef.current = "idle";
      setStatus("idle");
    }
  }, []);

  const run = useCallback(async () => {
    // No-op if already running
    if (statusRef.current === "running") return;

    // Debounce: if test completed within the last 60 seconds, return immediately
    const now = Date.now();
    if (now - lastCompletedRef.current < DEBOUNCE_MS) {
      return;
    }

    if (controllerRef.current) {
      controllerRef.current.abort();
    }

    const controller = new AbortController();
    controllerRef.current = controller;

    statusRef.current = "running";
    setStatus("running");
    setError(null);

    const url = `https://speed.cloudflare.com/__down?bytes=1000000&_t=${now}`;
    const start = performance.now();

    try {
      const res = await fetch(url, {
        cache: "no-store",
        signal: controller.signal,
      });

      if (!res.ok) {
        throw new Error(`HTTP ${res.status}`);
      }

      const blob = await res.blob();
      const end = performance.now();
      const durationMs = Math.max(1, end - start);
      const mbps =
        Math.round(((blob.size * 8) / (durationMs / 1000) / 1_000_000) * 10) /
        10;

      if (!isMountedRef.current) return;

      lastCompletedRef.current = Date.now();
      const newResult = {
        mbps,
        bytes: blob.size,
        durationMs: Math.round(durationMs),
        timestamp: lastCompletedRef.current,
      };

      resultRef.current = newResult;
      setResult(newResult);
      statusRef.current = "done";
      setStatus("done");
    } catch (err) {
      if (!isMountedRef.current) return;

      if (err?.name === "AbortError") {
        // Silent abort: clear running state
        statusRef.current = resultRef.current ? "done" : "idle";
        setStatus(statusRef.current);
        return;
      }

      setError(err?.message || "Speed test failed");
      statusRef.current = "error";
      setStatus("error");
    } finally {
      if (controllerRef.current === controller) {
        controllerRef.current = null;
      }
    }
  }, []);

  return {
    run,
    result,
    status,
    error,
    reset,
    abort,
  };
}

// Stable "frequency" from station name / _id (cosmetic, not real signal data)

function hashToFreq(str) {
  let h = 0;
  for (let i = 0; i < str.length; i++) {
    h = (h * 31 + str.charCodeAt(i)) | 0;
  }
  const norm = ((h >>> 0) % 2001) / 100; // 0.00 - 20.00
  return 88.0 + norm; // 88.0 - 108.0
}

function getFrequency(station) {
  if (station?.frequency) return Number(station.frequency);
  const seed = station?._id || station?.name || "unknown";
  return Math.round(hashToFreq(seed) * 10) / 10;
}

// ---------------------------------------------------------------------------
// Equalizer
//
// Reads real frequency data from the analyser when the player has one.
// If the stream can't be analysed (no CORS headers) it falls back to a
// simulated animation and reports that via onMode, so the UI can say so.
// When nothing is playing, the loop settles and STOPS (no animation, no CPU).
// ---------------------------------------------------------------------------

const BAR_COUNT = 24;
const FLOOR = 3;

function makeBands(binCount) {
  const maxBin = Math.min(binCount - 1, Math.round(binCount * 0.55));
  const edges = [];
  for (let i = 0; i <= BAR_COUNT; i++) {
    edges.push(1 + Math.round(Math.pow(i / BAR_COUNT, 1.8) * (maxBin - 1)));
  }
  return Array.from({ length: BAR_COUNT }, (_, i) => [
    edges[i],
    Math.max(edges[i] + 1, edges[i + 1]),
  ]);
}

function bandLevels(data, bands) {
  return bands.map(([lo, hi], i) => {
    let peak = 0;
    for (let b = lo; b < hi && b < data.length; b++) {
      if (data[b] > peak) peak = data[b];
    }
    const tilt = 1 + (i / BAR_COUNT) * 0.7; // highs carry less energy
    return Math.min(1, Math.pow(peak / 255, 1.5) * tilt);
  });
}

function Equalizer({ active, analyserRef, color, onMode }) {
  const canvasRef = useRef(null);
  const heightsRef = useRef(new Array(BAR_COUNT).fill(FLOOR));
  const sizeRef = useRef({ width: 0, height: 0, dpr: 1 });

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");

    const paint = () => {
      const { width: w, height: h } = sizeRef.current;
      if (w === 0 || h === 0) return;
      ctx.clearRect(0, 0, w, h);

      const gap = 4;
      const barW = Math.max(3, (w - (BAR_COUNT - 1) * gap) / BAR_COUNT);
      ctx.fillStyle = color;

      heightsRef.current.forEach((bh, i) => {
        const hh = Math.min(bh, h);
        ctx.fillRect(
          Math.floor(i * (barW + gap)),
          Math.floor(h - hh),
          Math.floor(barW),
          Math.ceil(hh)
        );
      });
    };

    const updateSize = () => {
      const dpr = window.devicePixelRatio || 1;
      const w = canvas.offsetWidth;
      const h = canvas.offsetHeight;
      if (w === 0 || h === 0) return;
      const s = sizeRef.current;
      if (s.width !== w || s.height !== h || s.dpr !== dpr) {
        sizeRef.current = { width: w, height: h, dpr };
        canvas.width = Math.floor(w * dpr);
        canvas.height = Math.floor(h * dpr);
        ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
        paint(); // resizing clears the canvas; redraw even if the loop is idle
      }
    };

    updateSize();

    let observer;
    if (typeof ResizeObserver !== "undefined") {
      observer = new ResizeObserver(updateSize);
      observer.observe(canvas);
    }

    let raf = 0;
    let data = null;
    let bands = null;
    let silentFrames = 0;
    let heard = false;
    let mode = null;

    const setMode = (m) => {
      if (m !== mode) {
        mode = m;
        onMode(m);
      }
    };

    const render = (time) => {
      const { height: h } = sizeRef.current;
      const analyser = active ? analyserRef.current : null;
      let levels = null;

      if (analyser) {
        if (!data || data.length !== analyser.frequencyBinCount) {
          data = new Uint8Array(analyser.frequencyBinCount);
          bands = makeBands(data.length);
        }
        analyser.getByteFrequencyData(data);

        let peak = 0;
        for (let i = 0; i < data.length; i++) {
          if (data[i] > peak) peak = data[i];
        }
        if (peak > 0) {
          heard = true;
          silentFrames = 0;
        } else {
          silentFrames++;
        }

        // Analyser present but never produced a single non-zero sample in
        // ~3s: the browser is feeding it silence. Treat as unavailable.
        if (heard || silentFrames < 180) levels = bandLevels(data, bands);
      }

      setMode(levels ? "live" : active ? "sim" : "idle");

      const heights = heightsRef.current;
      let moving = false;

      for (let i = 0; i < BAR_COUNT; i++) {
        let target = FLOOR;

        if (active) {
          if (levels) {
            target = Math.max(FLOOR, levels[i] * h * 0.95);
          } else {
            const t = time * 0.0012;
            const a = Math.sin(t * 1.5 + i * 0.4);
            const b = Math.cos(t * 2.1 - i * 0.3);
            const c = Math.sin(t * 3.3 + i * 0.7);
            const norm = (a * 0.4 + b * 0.4 + c * 0.2 + 1) / 2;
            target = Math.max(6, norm * (h * 0.85));
          }
        }

        // fast attack so beats hit, slower release so bars fall smoothly
        const diff = target - heights[i];
        heights[i] += diff * (diff > 0 ? 0.6 : 0.15);
        if (heights[i] > FLOOR + 0.5) moving = true;
      }

      paint();

      // Keep looping only while playing or while bars are still falling.
      if (active || moving) raf = requestAnimationFrame(render);
    };

    raf = requestAnimationFrame(render);

    return () => {
      cancelAnimationFrame(raf);
      if (observer) observer.disconnect();
    };
  }, [active, analyserRef, color, onMode]);

  return (
    <div className="relative min-h-[48px] flex-1">
      <canvas
        ref={canvasRef}
        className="absolute inset-0 block h-full w-full"
      />
    </div>
  );
}

// One colour per genre. Known genres get a fixed colour; anything else gets a
// stable colour hashed from its name, so new genres from the API still work.
// Every colour is bright, so black text is the readable choice on top of it.
//
// The palette below was generated by walking the hue wheel in even ~13° steps
// and alternating lightness so that even hue-neighbours read as different
// colours. Saturation is kept high throughout — this is a shouting palette.

const KNOWN = {
  // ---- reds (hue 0–40) ----
  metal: "#ff0000", // 0°   pure red
  punk: "#ff2200", // 13°  red-orange
  latin: "#ff4400", // 26°  burnt orange
  christian: "#0f7476",

  // ---- oranges (hue 40–70) ----
  rock: "#ff6600", // 39°  orange
  variety: "#ff8800", // 52°  light orange
  religious: "#ffaa00", // 65°  amber

  // ---- yellows (hue 70–110) ----
  gospel: "#ffcc00", // 78°  golden yellow
  jazz: "#ddee00", // 91°  olive yellow

// Stable palette for genres the API returns that aren't in KNOWN. Every entry
// is a distinct darker shade in a family that no KNOWN entry uses — so an
// unknown genre never collides with a known one.
const FALLBACK = [
  "#cc0000", // dark red
  "#cc6600", // brown-orange
  "#cccc00", // mustard
  "#66cc00", // grass green
  "#00cc66", // emerald
  "#00cccc", // turquoise
  "#0066cc", // steel blue
  "#6600cc", // royal purple
  "#cc00cc", // orchid
  "#cc0066", // raspberry
];

export function getGenreColor(genre) {
  const key = (genre || "unknown").trim().toLowerCase();
  if (KNOWN[key]) return KNOWN[key];

  let h = 0;
  for (let i = 0; i < key.length; i++) {
    h = (h * 31 + key.charCodeAt(i)) | 0;
  }
  return FALLBACK[(h >>> 0) % FALLBACK.length];
}

// Black or white, whichever reads better on the given #rrggbb colour.
export function readableOn(hex) {
  const n = parseInt(hex.slice(1), 16);
  const r = (n >> 16) & 255;
  const g = (n >> 8) & 255;
  const b = n & 255;
  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
  return luminance > 0.55 ? "#000000" : "#ffffff";
}
